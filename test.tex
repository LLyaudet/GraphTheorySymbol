\newcommand{\GTStestText}[2]{
\noindent

Abbreviation: #1B., B.#1, B.#1B.\newline

Index math : normal $G_{a#1b}$, relation $G_{u#2v}$.\newline
Exponent math : normal $H^{a#1b}$, relation $H^{u#2v}$.\newline
}
